function persistSessionContext(context) {
    if (!context || !context.token) {
        localStorage.removeItem('redytelca_session');
        return;
    }
    localStorage.setItem('redytelca_session', JSON.stringify(context));
}

function renderMenu() {
    const container = document.getElementById('sidebar-nav');
    if (!container) {
        return;
    }

    const normalizePageId = (value) => {
        const raw = String(value || '').trim().toLowerCase();
        const aliases = {
            'dashboard': 'dash',
            'inicio': 'dash',
            'gestión de clientes': 'clientes',
            'gestion de clientes': 'clientes',
            'registrar cliente': 'registro',
            'control de tareas': 'tareas',
            'cambiar contraseña': 'password',
            'configuración': 'configuracion',
            'configuracion': 'configuracion',
            'administración y accesos': 'configuracion',
            'administracion y accesos': 'configuracion',
            'permisos': 'permisos',
            'usuarios': 'usuarios',
            'tickets': 'tickets',
            'mapa': 'mapa',
            'reportes': 'reportes'
        };
        return aliases[raw] || raw || 'dash';
    };

    const items = [...buildDefaultMenuItems()];
    const seen = new Map(items.map(item => [item.page, item]));

    (window.permissionContext.modules || []).forEach(entry => {
        const pageId = normalizePageId(entry.url || entry.slug || entry.page || entry.vista_key || entry.pagina || entry.vista || entry.modulo);
        if (!pageId || seen.has(pageId)) {
            return;
        }

        items.push({
            page: pageId,
            label: entry.vista || entry.pagina || entry.label || entry.modulo || entry.url || entry.slug || 'Vista',
            module: entry.modulo || 'Principal'
        });
        seen.set(pageId, true);
    });

    const groups = {};
    items.forEach(item => {
        if (!groups[item.module]) {
            groups[item.module] = [];
        }
        groups[item.module].push(item);
    });

    const groupNames = Object.keys(groups);
    container.innerHTML = groupNames.map(groupName => {
        const itemsMarkup = groups[groupName].map(item => `
            <div class="nav-item" data-page="${item.page}" onclick="showPage('${item.page}', this)">${item.label}</div>
        `).join('');

        return `
            <div class="nav-group">
                <div class="nav-group-title">${groupName}</div>
                ${itemsMarkup}
            </div>
        `;
    }).join('');

    const firstItem = container.querySelector('.nav-item');
    if (firstItem) {
        showPage(firstItem.getAttribute('data-page'), firstItem);
    }
}

async function restoreSession() {
    const stored = getStoredSession();
    if (!stored || !stored.token) {
        clearSessionContext();
        return false;
    }

    try {
        const response = await fetch('auth_state.php', {
            headers: { 'X-Session-Token': stored.token }
        });
        const result = await response.json();

        if (!result || result.status !== 'success' || !result.authenticated) {
            clearSessionContext();
            return false;
        }

        window.permissionContext = result;
        persistSessionContext(result);

        const label = document.getElementById('profile-name-label');
        if (label) {
            label.textContent = result.user || 'Usuario';
        }

        renderMenu();
        return true;
    } catch (error) {
        clearSessionContext();
        return false;
    }
}

function escapeHtml(value) {
    return String(value ?? '')
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
}

        if (priority) priority.value = 'Media';
        if (client) client.value = '';
        if (user) user.value = '';
        if (id) id.value = '';
    }
}

async function guardarTarea() {
    const id = document.getElementById('task-id').value;
    const payload = {
        titulo: document.getElementById('task-title').value.trim(),

    };

function getStatusClass(status) {
    const value = String(status || '').toLowerCase();
    if (value.includes('complet') || value.includes('cerr')) return 'status-ok';
    if (value.includes('curso') || value.includes('proceso')) return 'status-warning';
    return 'status-alert';
}

function setMapFilter(filter, element) {
    const buttons = document.querySelectorAll('#map-filters .filter-pill');
    buttons.forEach(btn => btn.classList.remove('active'));
    if (element) {
        element.classList.add('active');
    }
    const markers = document.querySelectorAll('#mapa .map-marker');
    markers.forEach(marker => {
        const matches = filter === 'Todos' || marker.getAttribute('data-type') === filter;
        marker.style.opacity = matches ? '1' : '0.3';
    });
}

function setMapMode(mode, element) {
    const panel = document.getElementById('map-panel');
    const buttons = document.querySelectorAll('.view-toggle .toggle-btn');
    buttons.forEach(btn => btn.classList.remove('active'));
    if (panel) {
        panel.classList.toggle('satellite', mode === 'satelite');
    }
    if (element) {
        element.classList.add('active');
    }
}

function updateDashboardData() {
    const date = new Date();
    const formattedDate = date.toLocaleDateString('es-ES', {
        day: '2-digit',
        month: 'long',
        year: 'numeric'
    });

    const dashboardDate = document.getElementById('dashboard-date');
    if (dashboardDate) {
        dashboardDate.textContent = formattedDate;
    }

    const clientsCount = document.getElementById('stat-clients');
    if (clientsCount) {
        clientsCount.textContent = window.clientesCache?.length || 0;
    }

    const totalClientes = window.clientesCache?.length || 0;
    const ticketsAbiertos = Math.max(0, 18 + (totalClientes % 5) - 2);

    const ticketsValue = document.querySelector('.stat-card.border-yellow h2');
    if (ticketsValue) {
        ticketsValue.textContent = ticketsAbiertos;
    }

    const servicePercent = Math.max(70, Math.min(99, 78 + (totalClientes % 8)));
    const servicesValue = document.querySelector('.donut-center strong');
    if (servicesValue) {
        servicesValue.textContent = `${servicePercent}%`;
    }

    const oltSanFrancisco = document.getElementById('olt-san-francisco');
    const oltElBajo = document.getElementById('olt-el-bajo');
    const oltElGuacharo = document.getElementById('olt-el-guacharo');

    if (oltSanFrancisco || oltElBajo || oltElGuacharo) {
        const counts = {
            'san francisco': 0,
            'el bajo': 0,
            'el guacharo': 0
        };

        window.clientesCache.forEach(cliente => {
            const text = `${cliente.direccion || ''} ${cliente.olt || ''}`.toLowerCase();
            if (text.includes('san francisco')) counts['san francisco']++;
            else if (text.includes('el bajo')) counts['el bajo']++;
            else if (text.includes('guacharo') || text.includes('guácharo')) counts['el guacharo']++;
        });

        if (oltSanFrancisco) oltSanFrancisco.textContent = counts['san francisco'];
        if (oltElBajo) oltElBajo.textContent = counts['el bajo'];
        if (oltElGuacharo) oltElGuacharo.textContent = counts['el guacharo'];
    }

    const panel = document.getElementById('client-profile-panel');
    if (panel && panel.style.display !== 'none' && panel.dataset.clientId) {
        const currentClient = window.clientesCache.find(c => c.id_cliente === Number(panel.dataset.clientId));
        if (currentClient) {
            renderClientProfile(currentClient);
        }
    }
}

function renderReportes() {
    const totalClientesEl = document.getElementById('report-total-clientes');
    const tareasPendientesEl = document.getElementById('report-tareas-pendientes');
    const tareasCompletadasEl = document.getElementById('report-tareas-completadas');
    const ticketsAbiertosEl = document.getElementById('report-tickets-abiertos');
    const ticketsPrioridadEl = document.getElementById('report-tickets-prioridad');

    if (!totalClientesEl && !tareasPendientesEl && !ticketsAbiertosEl) {
        return;
    }

    const tareas = Array.isArray(window.tareasCache) ? window.tareasCache : [];
    const tickets = Array.isArray(window.ticketsCache) ? window.ticketsCache : [];
    const clientesTotal = Number(window.clientesCache?.length || 0);

    const tareasPendientes = tareas.filter(t => String(t.estado || '').toLowerCase() !== 'completada').length;
    const tareasCompletadas = tareas.filter(t => String(t.estado || '').toLowerCase() === 'completada').length;
    const ticketsAbiertos = tickets.filter(t => String(t.estado || '').toLowerCase() !== 'cerrado').length;

    if (totalClientesEl) totalClientesEl.textContent = clientesTotal;
    if (tareasPendientesEl) tareasPendientesEl.textContent = tareasPendientes;
    if (tareasCompletadasEl) tareasCompletadasEl.textContent = tareasCompletadas;
    if (ticketsAbiertosEl) ticketsAbiertosEl.textContent = ticketsAbiertos;

    if (ticketsPrioridadEl) {
        const porPrioridad = { Alta: 0, Media: 0, Baja: 0 };
        tickets.forEach(t => {
            const prioridad = t.prioridad && porPrioridad.hasOwnProperty(t.prioridad) ? t.prioridad : 'Media';
            porPrioridad[prioridad]++;
        });
        ticketsPrioridadEl.innerHTML = Object.keys(porPrioridad).map(prioridad => `
            <div><span>Prioridad ${escapeHtml(prioridad)}</span><strong>${porPrioridad[prioridad]}</strong></div>
        `).join('');
    }
}

function renderClientProfile(cliente) {
    const panel = document.getElementById('client-profile-panel');
    if (!panel) {
        return;
    }

    panel.style.display = 'block';
    panel.dataset.clientId = cliente.id_cliente;

    const zoneSelect = document.getElementById('profile-zone');
    if (zoneSelect) {
        const zoneValue = cliente.zona || cliente.olt || 'BAJO';
        zoneSelect.value = ['BAJO', 'VILLA', 'FIBRA-VILLA', 'PRU', 'OTRO'].includes(zoneValue)
            ? zoneValue
            : 'OTRO';
    }

    const nameEl = document.getElementById('profile-name');
    const statusEl = document.getElementById('profile-status-badge');
    const stateEl = document.getElementById('profile-state');
    const routerEl = document.getElementById('profile-router');
    const idEl = document.getElementById('profile-id');
    const addressEl = document.getElementById('profile-address');
    const phoneEl = document.getElementById('profile-phone');
    const emailEl = document.getElementById('profile-email');
    const paymentDaysEl = document.getElementById('notify-payment-days');
    const invoiceEl = document.getElementById('notify-invoice');
    const screenEl = document.getElementById('notify-screen');
    const smsEl = document.getElementById('notify-sms');
    const nextCutEl = document.getElementById('payment-next-cut');
    const statusFinanceEl = document.getElementById('payment-status');
    const debtEl = document.getElementById('payment-debt');
    const balanceEl = document.getElementById('payment-balance');

    if (nameEl) nameEl.textContent = `${cliente.nombres || ''} ${cliente.apellidos || ''}`.trim();
    if (statusEl) {
        statusEl.textContent = (cliente.estado || 'Activo').toUpperCase();
        statusEl.className = 'status ' + ((cliente.estado || 'Activo') === 'Inactivo' ? 'status-warn' : 'status-ok');
    }
    if (stateEl) stateEl.textContent = cliente.estado || 'Activo';
    if (routerEl) routerEl.textContent = cliente.olt || 'Sin asignar';
    if (idEl) idEl.textContent = cliente.cedula || '-';
    if (addressEl) addressEl.textContent = cliente.direccion || '-';
    if (phoneEl) phoneEl.textContent = cliente.num_telefono || '-';
    if (emailEl) emailEl.textContent = cliente.correo || '-';

    if (paymentDaysEl) paymentDaysEl.textContent = cliente.dias_pago || '15';
    if (invoiceEl) invoiceEl.textContent = cliente.factura_emitida ? 'Emitida' : 'Pendiente';
    if (screenEl) screenEl.textContent = cliente.aviso_pantalla ? 'Sí' : 'No';
    if (smsEl) smsEl.textContent = cliente.aviso_sms ? 'Sí' : 'No';
    if (nextCutEl) nextCutEl.textContent = cliente.proximo_corte || '25/06';
    if (statusFinanceEl) statusFinanceEl.textContent = cliente.estado_financiero || 'Al día';
    if (debtEl) debtEl.textContent = cliente.deuda ? `$${Number(cliente.deuda).toFixed(2)}` : '$0,00';
    if (balanceEl) balanceEl.textContent = cliente.saldo ? `$${Number(cliente.saldo).toFixed(2)}` : '$120,00';

    updateZoneLabel();
}

async function loadClientes(filter = '', page = 1) {
    const list = document.getElementById('clientes-list');
    const pagination = document.getElementById('clientes-pagination');
    if (!list) {
        return;
    }

    window.currentClienteSearch = typeof filter === 'string' ? filter : '';
    const safePage = Number(page) > 1 ? Number(page) : 1;

    try {
        const response = await fetch('clientes_listar.php?filter=' + encodeURIComponent(window.currentClienteSearch) + '&page=' + safePage + '&per_page=10');
        if (!response.ok) {
            throw new Error('Error al consultar clientes');
        }

        const result = await response.json();

        if (!result || result.status !== 'success' || !Array.isArray(result.clientes)) {
            throw new Error('Respuesta inválida del servidor');
        }

        const clientes = result.clientes;
        window.clientesCache = clientes;
        populateAssignmentSelectors();
        updateDashboardData();

        if (clientes.length === 0) {
            list.innerHTML = '<p>No hay clientes registrados.</p>';
            if (pagination) {
                pagination.innerHTML = '';
            }
            return;
        }

        list.innerHTML = clientes.map(cliente => `
            <div class="card" style="margin-bottom: 8px;">
                <p><strong>${cliente.nombres || ''} ${cliente.apellidos || ''}</strong> - Cédula: ${cliente.cedula || ''}</p>
                <p><small>Teléfono: ${cliente.num_telefono || ''} · Correo: ${cliente.correo || ''}</small></p>
                <p><small>Dirección: ${cliente.direccion || 'No registrada'} · OLT: ${cliente.olt || 'Sin asignar'}</small></p>
                <div style="display:flex; gap:8px; flex-wrap:wrap;">
                    <button data-client-action="profile" data-client-id="${cliente.id_cliente}" style="width:auto; padding:10px 14px;">Ver ficha</button>
                    <button data-client-action="edit" data-client-id="${cliente.id_cliente}" style="width:auto; padding:10px 14px;">Editar</button>
                    <button data-client-action="delete" data-client-id="${cliente.id_cliente}" style="width:auto; padding:10px 14px; background:#dc2626;">Eliminar</button>
                </div>
            </div>
        `).join('');

        if (pagination) {
            const totalPages = Number(result.pages || 1);
            const currentPage = Number(result.page || 1);
            pagination.innerHTML = `
                <div class="pagination-controls">
                    <button onclick="changeClientPage(${Math.max(1, currentPage - 1)})" ${currentPage <= 1 ? 'disabled' : ''}>Anterior</button>
                    <span>Página ${currentPage} / ${totalPages} · ${result.total || clientes.length} clientes</span>
                    <button onclick="changeClientPage(${Math.min(totalPages, currentPage + 1)})" ${currentPage >= totalPages ? 'disabled' : ''}>Siguiente</button>
                </div>
            `;
        }

        if (clientes.length > 0) {
            renderClientProfile(clientes[0]);
        }
    } catch (error) {
        console.warn('Usando datos de respaldo para clientes:', error);
        window.clientesCache = fallbackClientes;
        populateAssignmentSelectors();
        updateDashboardData();

        list.innerHTML = fallbackClientes.map(cliente => `
            <div class="card" style="margin-bottom: 8px;">
                <p><strong>${cliente.nombres} ${cliente.apellidos}</strong> - Cédula: ${cliente.cedula}</p>
                <p><small>Teléfono: ${cliente.num_telefono} · Correo: ${cliente.correo}</small></p>
                <p><small>Dirección: ${cliente.direccion || 'No registrada'} · OLT: ${cliente.olt || 'Sin asignar'}</small></p>
                <div style="display:flex; gap:8px; flex-wrap:wrap;">
                    <button data-client-action="profile" data-client-id="${cliente.id_cliente}" style="width:auto; padding:10px 14px;">Ver ficha</button>
                    <button data-client-action="edit" data-client-id="${cliente.id_cliente}" style="width:auto; padding:10px 14px;">Editar</button>
                    <button data-client-action="delete" data-client-id="${cliente.id_cliente}" style="width:auto; padding:10px 14px; background:#dc2626;">Eliminar</button>
                </div>
            </div>
        `).join('');

        if (pagination) {
            pagination.innerHTML = '';
        }

        renderClientProfile(fallbackClientes[0]);
    }
}

async function loadRoles() {
    const select = document.getElementById('role-select');
    const createSelect = document.getElementById('new-user-role');
    const configSelect = document.getElementById('config-role-select');

    try {
        const response = await fetch('roles_listar.php');
        const text = await response.text();
        let result;

        try {
            result = JSON.parse(text);
        } catch (error) {
            result = null;
        }

        const roles = (result && Array.isArray(result.roles))
            ? result.roles
            : [
                { id_rol: 1, nombre_rol: 'Administrador' },
                { id_rol: 2, nombre_rol: 'Operador' }
            ];

        const options = roles.map(rol => `<option value="${rol.id_rol}">${escapeHtml(rol.nombre_rol)}</option>`).join('');

        if (select) select.innerHTML = options;
        if (createSelect) createSelect.innerHTML = options;
        if (configSelect) configSelect.innerHTML = options;

        if (roles.length > 0 && select) {
            loadRolePermissions();
        }
        if (roles.length > 0 && configSelect) {
            loadRolePermissions('config');
        }
    } catch (error) {
        if (select) select.innerHTML = '<option value="1">Administrador</option>';
        if (createSelect) createSelect.innerHTML = '<option value="1">Administrador</option>';
        if (configSelect) configSelect.innerHTML = '<option value="1">Administrador</option>';
        console.error(error);
    }
}

async function crearRol() {

    const feedback = document.getElementById('config-role-feedback');

        if (feedback) {

        }
        return;
    }

    try {
        const response = await fetch('roles_listar.php', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ nombre_rol: nombreRol })
        });
        const result = await response.json();

        if (feedback) {
            feedback.style.color = result.status === 'success' ? 'green' : 'red';

async function loadRolePermissions(target = 'default') {
    const selectId = target === 'config' ? 'config-role-select' : 'role-select';
    const containerId = target === 'config' ? 'config-permissions-list' : 'permissions-list';
    const select = document.getElementById(selectId);
    const container = document.getElementById(containerId);

    if (!select || !container) {

            feedback.style.color = result.status === 'success' ? 'green' : 'red';

async function login() {
    const user = document.getElementById('user').value.trim();
    const pass = document.getElementById('pass').value.trim();

    if (!user || !pass) {
        alert('Por favor ingresa usuario y contrasena.');
        return;
    }

    try {
        const response = await fetch('login.php', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ username: user, password: pass })
        });

        if (!response.ok) {
            const text = await response.text();
            throw new Error('Error de servidor: ' + response.status + ' - ' + text);
        }

        const result = await response.json();
        if (result.status === 'success') {
            window.permissionContext = {
                authenticated: true,
                user: result.usuario,
                roleId: result.id_rol,
                roleName: result.rol_nombre || 'Rol',
                token: result.token || '',
                modules: result.modulos_permitidos || []
            };
            persistSessionContext(window.permissionContext);
            document.getElementById('login-overlay').style.display = 'none';
            document.getElementById('user').value = '';
            document.getElementById('pass').value = '';
            const label = document.getElementById('profile-name-label');
            if (label) {
                label.textContent = result.usuario || 'Usuario';
            }
            renderMenu();
            await loadRoles();
            await loadUsers();
            await loadClientes();
        } else {
            alert(result.message);
        }
    } catch (error) {
        console.error(error);
        alert('Error en el login: ' + error.message);
    }
}

header('Content-Type: application/json; charset=utf-8');

try {
    $pdo->exec("ALTER TABLE clientes ADD COLUMN IF NOT EXISTS direccion VARCHAR(255) NOT NULL DEFAULT ''");
    $pdo->exec("ALTER TABLE clientes ADD COLUMN IF NOT EXISTS olt VARCHAR(100) NOT NULL DEFAULT ''");
} catch (Exception $e) {
    // Ignorar si la tabla ya tiene esas columnas o si la base no soporta la sintaxis exacta.
}

$filter = isset($_GET['filter']) ? trim($_GET['filter']) : '';

$perPage = isset($_GET['per_page']) ? intval($_GET['per_page']) : 10;
$perPage = max(1, min(100, $perPage)); // límite de seguridad, evita per_page=999999

$page = isset($_GET['page']) ? intval($_GET['page']) : 1;
$page = max(1, $page);

$whereSql = '';
$params = [];
if ($filter !== '') {
    $whereSql = " WHERE nombres LIKE ? OR apellidos LIKE ? OR cedula LIKE ? OR correo LIKE ? OR COALESCE(direccion, '') LIKE ? OR COALESCE(olt, '') LIKE ?";
    $filterParam = "%$filter%";
    $params = [$filterParam, $filterParam, $filterParam, $filterParam, $filterParam, $filterParam];
}

// 1) Contar el total ANTES de traer filas: permite calcular cuántas páginas
// existen sin cargar los 2000 registros en memoria.
$countQuery = "SELECT COUNT(*) FROM clientes" . $whereSql;
$stmtCount = $pdo->prepare($countQuery);
$stmtCount->execute($params);
$total = (int) $stmtCount->fetchColumn();
$totalPages = max(1, (int) ceil($total / $perPage));

if ($page > $totalPages) {
    $page = $totalPages;
}
$offset = ($page - 1) * $perPage;

// 2) Traer únicamente el bloque de la página solicitada (LIMIT/OFFSET)
$query = "SELECT id_cliente, nombres, apellidos, cedula, num_telefono, correo,
                 COALESCE(direccion, '') AS direccion, COALESCE(olt, '') AS olt, fecha_registro
          FROM clientes" . $whereSql . " ORDER BY fecha_registro DESC LIMIT ? OFFSET ?";

$stmt = $pdo->prepare($query);
$paramIndex = 1;
foreach ($params as $param) {
    $stmt->bindValue($paramIndex, $param, PDO::PARAM_STR);
    $paramIndex++;
}
$stmt->bindValue($paramIndex++, $perPage, PDO::PARAM_INT);
$stmt->bindValue($paramIndex++, $offset, PDO::PARAM_INT);
$stmt->execute();
$clientes = $stmt->fetchAll(PDO::FETCH_ASSOC);

echo json_encode([
    'status' => 'success',
    'clientes' => $clientes,
    'page' => $page,
    'pages' => $totalPages,
    'total' => $total,
    'per_page' => $perPage
]);

- Navegación: cada sección debe mapearse a rutas y permisos; el menú debe mostrar solo las opciones permitidas por el rol del usuario.

2) Gestión de la empresa
- Datos básicos: almacenar logo, nombre legal, colores corporativos y otros metadatos en la sección de Configuraciones.
- Uso: estos valores deben aplicarse dinámicamente en la interfaz (cabeceras, favicon, paleta de estilos) y ser editables por administradores.

3) Administración de usuarios y roles
- Roles: permitir la creación de roles (ej.: Administrador, Operador) con nombres y descripciones.
- Permisos: asignar permisos granulares por módulo y vista. Un permiso determina acceso a rutas, botones y operaciones (leer, crear, actualizar, eliminar).
- Asignación: usuarios vinculados a uno o varios roles; el sistema resuelve permisos efectivos por la unión de permisos de sus roles.

4) Diseño de la interfaz (Grid y paginación)
- Grid: implementar componentes tipo "Grid" para listar entidades (clientes, usuarios, roles) con columnas configurables.
- Paginación: incluir paginación y/o carga perezosa (infinite scroll) para manejar grandes volúmenes; soportar filtros y ordenamiento del lado del servidor.
- Accesibilidad: asegurar tablas responsivas y con soporte para teclado y lectores de pantalla.

5) Gestión de base de datos
- Modelo: documentar el diagrama Entidad-Relación y las tablas principales: usuarios, roles, permisos, rol_permiso, usuario_rol, clientes, etc.
- Relaciones: definir claves foráneas y cardinalidades; mantener integridad referencial y restricciones apropiadas.
- Permisos: almacenar permisos en tablas normalizadas para permitir consultas eficientes sobre acceso por usuario/rol.

6) Seguridad y Sesiones
- Autenticación: usar tokens (JWT o similares) para sesiones; el token debe tener expiración y firmarse con clave segura.
- Almacenamiento: utilizar Local Storage o Session Storage según el caso, pero limpiar tokens y datos sensibles al cerrar sesión.
- Renovación y revocación: soportar refresh tokens y mecanismos para revocar sesiones (ej.: lista negra o control de versiones de token).
- Buenas prácticas: transmitir siempre por HTTPS, validar permisos en servidor, y evitar exponer datos sensibles en el cliente.

Notas finales

- Recomendación: incluir ejemplos de API y un esquema JSON de permisos para facilitar integraciones.

        INNER JOIN modulos m ON rmp.id_modulo = m.id_modulo
        INNER JOIN paginas p ON rmp.id_pagina = p.id_pagina
        WHERE rmp.id_rol = ?
        ORDER BY m.id_modulo, p.id_pagina");
    $stmt->execute([$roleId]);
    $rows = $stmt->fetchAll(PDO::FETCH_ASSOC);

    return array_map(fn($row) => [
        'modulo' => $row['modulo'],
        'vista' => $row['vista'],
        'url' => $row['url']
    ], $rows);

    // se reinserta desde cero con lo que llegó marcado en la interfaz.
    $stmtDel = $pdo->prepare("DELETE FROM rol_modulo_pagina WHERE id_rol = ?");

            }
            $stmtInsert->execute([$id_rol, (int) $idModulo, $idPagina]);
        }

<?php
header('Content-Type: application/json; charset=utf-8');
require 'conexion.php';

$input = file_get_contents('php://input');
$data = json_decode($input, true);
if (!$data) {
    $data = $_POST;
}

$username = trim($data['username'] ?? '');
$password = trim($data['password'] ?? '');
$idRol = isset($data['id_rol']) ? intval($data['id_rol']) : 0;

if (!$username || !$password || $idRol <= 0) {
    echo json_encode(['status' => 'error', 'message' => 'Todos los campos son obligatorios']);
    exit;
}

$stmt = $pdo->prepare('SELECT id_usuario FROM usuarios WHERE username = ?');
$stmt->execute([$username]);
if ($stmt->fetch()) {
    echo json_encode(['status' => 'error', 'message' => 'Ese usuario ya existe']);
    exit;
}

$stmt = $pdo->prepare('INSERT INTO usuarios (username, password, id_rol) VALUES (?, ?, ?)');
$ok = $stmt->execute([$username, $password, $idRol]);

echo json_encode([
    'status' => $ok ? 'success' : 'error',
    'message' => $ok ? 'Usuario creado correctamente' : 'No se pudo crear el usuario'
]);
